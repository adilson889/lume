                <set name="posceu" value="posceu + 1"></set>
            </if>
            <else>
                <if condition="nome == 'casa' || nome == 'edifício' || nome == 'montanha' || nome == 'barco'">
                    <if condition="totalcentro <= 2"><set name="escala" value="0.78"></set></if>
                    <else><if condition="totalcentro <= 4"><set name="escala" value="0.62"></set></if><else><if condition="totalcentro <= 6"><set name="escala" value="0.48"></set></if><else><set name="escala" value="0.36"></set></else></else></else>
                    <if condition="totalcentro == 1"><set name="x" value="(325 - 100 * escala) / 2"></set></if>
                <else>
                    <set name="gap" value="(325 - 16 - totalcentro * 100 * escala) / (totalcentro - 1)"></set>
                    <if condition="gap > 6"><set name="gap" value="6"></set></if>
                    <set name="x" value="(325 - (totalcentro * 100 * escala + (totalcentro - 1) * gap)) / 2 + poscentro * (100 * escala + gap)"></set>
                </else>
                    <!-- Montanhas ficam no horizonte; casas e edifícios descem para o plano habitado. -->
                    <if condition="nome == 'montanha'">
                        <!-- Alinha a base da montanha exactamente com a linha do horizonte (y=165), independentemente da escala. -->

                    </if>
                    <else>
                        <if condition="nome == 'casa'">
                            <set name="y" value="112"></set><if condition="totalcentro > 4 && poscentro % 2 == 1"><set name="y" value="116"></set></if>
                        </if>
                        <else>
                            <if condition="nome == 'edifício'">
                                <set name="y" value="100"></set><if condition="totalcentro > 4 && poscentro % 2 == 1"><set name="y" value="104"></set></if>
                            </if>
                            <else>
                                <set name="y" value="130"></set><if condition="totalcentro > 4 && poscentro % 2 == 1"><set name="y" value="133"></set></if>
                            </else>
                        </else>
                    </else>
                    <set name="elementos" value="elementos + '<g transform=\"translate(' + x + ' ' + y + ') scale(' + escala + ')\">' + pecas_svg[p] + '</g>'"></set>
                    <set name="poscentro" value="poscentro + 1"></set>
                </if>
                <else>
                    <if condition="totalterra <= 2"><set name="escala" value="0.78"></set></if>
                    <else><if condition="totalterra <= 4"><set name="escala" value="0.62"></set></if><else><if condition="totalterra <= 6"><set name="escala" value="0.48"></set></if><else><set name="escala" value="0.36"></set></else></else></else>
                    <if condition="totalterra == 1"><set name="x" value="(325 - 100 * escala) / 2"></set></if>
                <else>
                    <set name="gap" value="(325 - 16 - totalterra * 100 * escala) / (totalterra - 1)"></set>

                    <set name="posterra" value="posterra + 1"></set>
                </else>
            </else>
        </for>
    </if>

<var name="total" value="posceu + poscentro + posterra"></var>

<else>
    <if condition="total == 0 && !includes(m, 'gráfico') && !includes(m, 'grafico')">
